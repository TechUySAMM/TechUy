<?php
require "conexion.php";

$errores = [];
$titulo = $tipo = $cedula = $fecha = $ruta = "";

if ($_SERVER["REQUEST_METHOD"] === "POST") {

    $titulo = trim($_POST["titulo"] ?? "");
    $tipo   = trim($_POST["tipo"] ?? "");
    $cedula = trim($_POST["cedula_paciente"] ?? "");
    $fecha  = trim($_POST["fecha_emision"] ?? "");
    $ruta   = trim($_POST["ruta_archivo"] ?? "");

    // 1) Título: obligatorio y máximo 150 caracteres
    //    (mb_strlen cuenta caracteres; strlen cuenta bytes y las tildes ocupan 2)
    if ($titulo === "") {
        $errores[] = "El título es obligatorio.";
    } elseif (mb_strlen($titulo) > 150) {
        $errores[] = "El título no puede superar los 150 caracteres.";
    }

    // 2) Tipo: solo "indicacion" o "informacion" (igual que el rol)
    if ($tipo !== "indicacion" && $tipo !== "informacion") {
        $errores[] = "Debe seleccionar un tipo válido.";
    }

    // 3) Cédula: 7 u 8 dígitos, sin puntos ni guion
    if (!preg_match('/^[0-9]{7,8}$/', $cedula)) {
        $errores[] = "La cédula debe tener 7 u 8 dígitos, sin puntos ni guion.";
    }

    // 4) Fecha real: "2026-09-28" -> [2026, 09, 28] -> checkdate(mes, día, año)
    $partes = explode("-", $fecha);
    if (count($partes) !== 3 || !checkdate((int)$partes[1], (int)$partes[2], (int)$partes[0])) {
        $errores[] = "La fecha de emisión no es válida.";
    }

    // 5) Ruta: empieza con documentos/ y termina en .pdf
    if (!preg_match('/^documentos\/[a-zA-Z0-9_-]+\.pdf$/', $ruta)) {
        $errores[] = "La ruta debe tener el formato documentos/nombre.pdf";
    }

    // Solo vamos a la base si no hubo errores
    if (empty($errores)) {
        $sql = "INSERT INTO documento
                    (titulo, tipo, cedula_paciente, fecha_emision, ruta_archivo, activo)
                VALUES
                    (:titulo, :tipo, :cedula, :fecha, :ruta, 1)";
        $stmt = $pdo->prepare($sql);

        try {
            $stmt->execute([
                ":titulo" => $titulo,
                ":tipo"   => $tipo,
                ":cedula" => $cedula,
                ":fecha"  => $fecha,
                ":ruta"   => $ruta
            ]);

            header("Location: documentos.php?ok=1");
            exit;

        } catch (PDOException $e) {
            // 23000 = viola una restricción: la ruta ya existe (UNIQUE)
            if ($e->getCode() === "23000") {
                $errores[] = "Ya existe un documento con esa ruta de archivo.";
            } else {
                throw $e;
            }
        }
    }
}
?>
<!DOCTYPE html>
<html lang="es" data-bs-theme="dark">
<head>
    <meta charset="utf-8">
    <meta name="viewport" content="width=device-width, initial-scale=1">
    <title>Nuevo documento - S.I.G.S.M.</title>
    <link href="https://cdn.jsdelivr.net/npm/bootstrap@5.3.3/dist/css/bootstrap.min.css" rel="stylesheet">
</head>
<body class="container py-4" style="max-width: 700px;">

    <h1 class="h4 mb-3">Nuevo documento</h1>

    <?php if (!empty($errores)): ?>
        <div class="alert alert-danger">
            <ul class="mb-0">
                <?php foreach ($errores as $e): ?>
                    <li><?= htmlspecialchars($e) ?></li>
                <?php endforeach; ?>
            </ul>
        </div>
    <?php endif; ?>

    <form method="POST" action="alta_documento.php">

        <div class="mb-3">
            <label class="form-label">Título</label>
            <input type="text" name="titulo" class="form-control"
                   value="<?= htmlspecialchars($titulo) ?>">
        </div>

        <div class="mb-3">
            <label class="form-label">Tipo</label>
            <select name="tipo" class="form-select">
                <option value="">Seleccione...</option>
                <option value="indicacion" <?= $tipo === "indicacion" ? "selected" : "" ?>>Indicación médica</option>
                <option value="informacion" <?= $tipo === "informacion" ? "selected" : "" ?>>Información general</option>
            </select>
        </div>

        <div class="mb-3">
            <label class="form-label">Cédula del paciente</label>
            <input type="text" name="cedula_paciente" class="form-control"
                   placeholder="45678912"
                   value="<?= htmlspecialchars($cedula) ?>">
            <div class="form-text">Sin puntos ni guion.</div>
        </div>

        <div class="mb-3">
            <label class="form-label">Fecha de emisión</label>
            <input type="date" name="fecha_emision" class="form-control"
                   value="<?= htmlspecialchars($fecha) ?>">
        </div>

        <div class="mb-3">
            <label class="form-label">Ruta del archivo (simulado)</label>
            <input type="text" name="ruta_archivo" class="form-control font-monospace"
                   placeholder="documentos/indicacion_45678912_001.pdf"
                   value="<?= htmlspecialchars($ruta) ?>">
        </div>

        <button type="submit" class="btn btn-primary">Guardar</button>
        <a href="documentos.php" class="btn btn-secondary">Cancelar</a>
    </form>

</body>
</html>
