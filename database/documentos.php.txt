<?php
require "conexion.php";

// Consulta fija (no hay datos del usuario), alcanza con query().
// Pedimos el id aunque no lo mostremos: lo necesitamos para Editar y Baja.
$sql = "SELECT id, titulo, tipo, cedula_paciente, fecha_emision, ruta_archivo
        FROM documento
        WHERE activo = 1
        ORDER BY fecha_emision DESC";
$stmt = $pdo->query($sql);
$documentos = $stmt->fetchAll();

// Mensaje de éxito según lo que llegue por ?ok=
$ok = $_GET["ok"] ?? "";
$mensaje = "";
if ($ok === "1") {
    $mensaje = "Documento creado correctamente.";
} elseif ($ok === "2") {
    $mensaje = "Documento modificado correctamente.";
} elseif ($ok === "3") {
    $mensaje = "Documento dado de baja correctamente.";
}
?>
<!DOCTYPE html>
<html lang="es" data-bs-theme="dark">
<head>
    <meta charset="utf-8">
    <meta name="viewport" content="width=device-width, initial-scale=1">
    <title>Documentos - S.I.G.S.M.</title>
    <link href="https://cdn.jsdelivr.net/npm/bootstrap@5.3.3/dist/css/bootstrap.min.css" rel="stylesheet">
</head>
<body class="container py-4" style="max-width: 1100px;">

    <div class="d-flex justify-content-between align-items-center mb-3">
        <h1 class="h4 mb-0">Documentos</h1>
        <a href="alta_documento.php" class="btn btn-primary btn-sm">+ Nuevo documento</a>
    </div>

    <?php if ($mensaje !== ""): ?>
        <div class="alert alert-success"><?= htmlspecialchars($mensaje) ?></div>
    <?php endif; ?>

    <?php if (empty($documentos)): ?>

        <div class="alert alert-secondary">Todavía no hay documentos cargados.</div>

    <?php else: ?>

        <p class="text-secondary">Total: <?= count($documentos) ?> documentos</p>

        <div class="table-responsive">
            <table class="table table-dark table-striped align-middle">
                <thead>
                    <tr>
                        <th>Título</th>
                        <th>Tipo</th>
                        <th>Cédula</th>
                        <th>Emisión</th>
                        <th>Archivo</th>
                        <th>Acciones</th>
                    </tr>
                </thead>
                <tbody>
                    <?php foreach ($documentos as $d): ?>
                        <tr>
                            <td><?= htmlspecialchars($d["titulo"]) ?></td>
                            <td><?= $d["tipo"] === "indicacion" ? "Indicación médica" : "Información general" ?></td>
                            <td><?= htmlspecialchars($d["cedula_paciente"]) ?></td>
                            <td><?= date("d/m/Y", strtotime($d["fecha_emision"])) ?></td>
                            <td><code><?= htmlspecialchars($d["ruta_archivo"]) ?></code></td>
                            <td class="text-nowrap">
                                <a href="editar_documento.php?id=<?= $d["id"] ?>"
                                   class="btn btn-sm btn-outline-warning">Editar</a>

                                <!-- La baja modifica datos: va por POST, nunca por un link -->
                                <form method="POST" action="baja_documento.php" class="d-inline"
                                      onsubmit="return confirm('¿Dar de baja este documento?');">
                                    <input type="hidden" name="id" value="<?= $d["id"] ?>">
                                    <button type="submit" class="btn btn-sm btn-outline-danger">Baja</button>
                                </form>
                            </td>
                        </tr>
                    <?php endforeach; ?>
                </tbody>
            </table>
        </div>

    <?php endif; ?>

</body>
</html>
